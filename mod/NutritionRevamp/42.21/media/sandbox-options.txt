/* Severity scales every deficiency and excess penalty of the effects table after the
   fold; 0 leaves only the health drains, which no option turns off; 1.0 is the default. */
option NR.Severity
{
    type = double, min = 0.0, max = 3.0, default = 1.0,
    page = NutritionRevamp, translation = NR_Severity,
}

/* VisibilityMode sets how much of the mod's state every player's
   interface shows: 1 Symptoms (the six symptom words and icons), 2 Bands, the default (and the grades and the body
   band), 3 Numbers (and every figure). A Nutritionist trait shows Numbers to its holder whatever the option. */
option NR.VisibilityMode
{
    type = enum, numValues = 3, default = 2,
    page = NutritionRevamp, translation = NR_VisibilityMode, valueTranslation = NR_VisibilityModeValues,
}

/* SexSpecific: off, every character uses the male values wherever the mod indexes by sex; on, a female
   character uses the female values. */
option NR.SexSpecific
{
    type = boolean, default = false,
    page = NutritionRevamp, translation = NR_SexSpecific,
}

/* StimulantRealism sets the stimulant sizes: 1 Game, the default (a cup's fatigue relief about twice vanilla's,
   intoxication about vanilla's per beer, a mild hangover), 2 Realistic (the pharmacological strengths). Absorption,
   clearance, tolerance, withdrawal and the hangover's start follow the science in both. */
option NR.StimulantRealism
{
    type = enum, numValues = 2, default = 1,
    page = NutritionRevamp, translation = NR_StimulantRealism, valueTranslation = NR_StimulantRealismValues,
}

/* DeathPace sets how fast starvation and dehydration take health once they begin: 1 Vanilla (vanilla's own level-4
   drain, a straight line at the server's day length), 2 Shortened (starvation in 14 game days, dehydration in 2, on
   the curve), 3 Realistic, the default (starvation at BMI 12, dehydration at a 17 percent water loss, on the curve). The
   deficiencies and the sodium and iron rungs are the same at every pace. */
option NR.DeathPace
{
    type = enum, numValues = 3, default = 3,
    page = NutritionRevamp, translation = NR_DeathPace, valueTranslation = NR_DeathPaceValues,
}

/* HealthCombine sets how the active lethal states' remaining fractions make the health target: 1 Worst only (the
   smallest), 2 Multiply, the default (their product), 3 Add (their losses summed, so several can kill before any one
   would). */
option NR.HealthCombine
{
    type = enum, numValues = 3, default = 2,
    page = NutritionRevamp, translation = NR_HealthCombine, valueTranslation = NR_HealthCombineValues,
}
